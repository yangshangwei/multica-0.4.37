SOURCE = ROOT.parent / 'mc-functional-go-7af9ab990-20261102'
TASK = ROOT / '.trellis/tasks/11-02-branch-functional-regression'
PRIVATE = Path(__file__).resolve().parent
RUN = TASK / 'evidence/next-webhook-lookup-1'

sha = lambda p: hashlib.sha256(Path(p).read_bytes()).hexdigest()
record = json.loads((RUN / 'run.json').read_text())

pg = {k: env[k] for k in ['PATH', 'LANG', 'LC_ALL'] if k in env}
pg.update(PGHOST=uri.hostname, PGPORT=str(uri.port), PGUSER=unquote(uri.username or ''),
          PGPASSWORD=unquote(uri.password or ''), PGDATABASE=DATABASE,
          PGCONNECT_TIMEOUT='5', PGSSLMODE=parse_qs(uri.query)['sslmode'][0],
          PGOPTIONS='-c default_transaction_read_only=on -c statement_timeout=5000')


def persist():
    (RUN / 'run.json').write_text(json.dumps(record, indent=2) + '\n')


def invoke(name, command, selected_env=env, timeout=600):
    step = dict(name=name, command=command, cwd=str(SOURCE), started_at=now())
    if (PRIVATE / 'diagnostic_test.go').exists():
        step['private_test_sha256'] = sha(PRIVATE / 'diagnostic_test.go')
    record['steps'].append(step)
    persist()
    with (RUN / (name + '.stdout')).open('wb') as out, (RUN / (name + '.stderr')).open('wb') as err:
        process = subprocess.Popen(command, cwd=SOURCE, env=selected_env, stdin=subprocess.DEVNULL,
                                   stdout=out, stderr=err, start_new_session=True)
        step.update(pid=process.pid, process_group=process.pid)
        persist()
        try:
            code = process.wait(timeout=timeout)
        except subprocess.TimeoutExpired:
            os.killpg(process.pid, signal.SIGTERM)
            process.wait(timeout=15)
            step['terminated_owned_group_for_timeout'] = True
            code = process.returncode
        step.update(exit_code=code, finished_at=now(), stdout_sha256=sha(RUN / (name + '.stdout')),
                    stderr_sha256=sha(RUN / (name + '.stderr')))
        persist()
        return code


def sql(name, query):
    code = invoke(name, [PSQL, '-X', '-At', '-v', 'ON_ERROR_STOP=1', '-c', query], pg, timeout=20)
    assert code == 0, 'Read-only task database check failed; see saved streams'
    value = (RUN / (name + '.stdout')).read_text().strip()
    with (RUN / 'database-operations.jsonl').open('a') as out:
        out.write(json.dumps(dict(at=now(), database=DATABASE, sql=query, exit_code=code,
                                  output_sha256=sha(RUN / (name + '.stdout')), read_only=True)) + '\n')
    return value


success = False
try:
