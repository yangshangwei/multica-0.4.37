def nodes(n):
    yield n
    for c in n.get('Plans',[]):yield from nodes(c)

def parse_value(v):
    if v in ['t','f']:return v=='t'
    if re.match(r'^\d{4}-\d\d-\d\d ',v):return datetime.datetime.fromisoformat(v.replace(' ','T')).isoformat()
    return v

def normalize_arg(v):
    if isinstance(v,str) and re.match(r'^\d{4}-\d\d-\d\dT',v):return datetime.datetime.fromisoformat(v.replace('Z','+00:00')).isoformat()
    return v
