def command(label, args, cwd=ROOT):
    started = NOW()
    p = subprocess.run(args, cwd=cwd, capture_output=True, timeout=30)
    (RUN / (label + '.stdout')).write_bytes(p.stdout)
    (RUN / (label + '.stderr')).write_bytes(p.stderr)
    result['commands'].append({'name': label, 'command': args, 'cwd': str(cwd),
                              'started_at': started, 'finished_at': NOW(),
                              'exit_code': p.returncode,
                              'stdout_sha256': SHA(p.stdout), 'stderr_sha256': SHA(p.stderr)})
    save()
    if p.returncode:
        raise RuntimeError(label + ' exit ' + str(p.returncode))
    return p.stdout


def filebytes(path):
    return os.readlink(path).encode() if path.is_symlink() else path.read_bytes()


def is_test(path):
    return (path.startswith('e2e/') or '/test/' in path or '/tests/' in path or
            path.endswith('_test.go') or '.test.' in path or '.spec.' in path or
            path.endswith('.test.sh') or '/__tests__/' in path)


def check_ledger(ledger):
    checked = []
    for name, expected in ledger.items():
        p = ROOT / name
        current = SHA(p.read_bytes()) if p.is_file() else None
        relative = str(p.relative_to(TASK)) if TASK in p.parents else name
        checked.append({'path': name, 'expected_sha256': expected, 'actual_sha256': current,
                        'exists': current is not None, 'same': current == expected,
                        'coordination_evolution_allowed': relative in COORDINATION})
    return checked
