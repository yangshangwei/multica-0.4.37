def command(name, args, expected=(0,)):
    step = {'name': name, 'command': args, 'started_at': now(), 'expected_exit_codes': list(expected)}
    record['steps'].append(step)
    save()
    executable = shutil.which(args[0])
    step['resolved_executable'] = executable
    if executable is None:
        step.update(status='NOT_RUN_MISSING_EXECUTABLE', finished_at=now())
        save()
        return None
    result = subprocess.run(args, cwd=ROOT, capture_output=True, timeout=30)
    for suffix, data in [('stdout', result.stdout), ('stderr', result.stderr)]:
        (RUN / (name + '.' + suffix)).write_bytes(data)

    step.update(exit_code=result.returncode, finished_at=now(), status='RECORDED' if result.returncode in expected else 'UNEXPECTED_EXIT_RECORDED')
    save()
    return result

def artifact(path, expected, role):
    p = Path(path)
    row = {'path': str(p), 'role': role, 'expected_sha256': expected, 'exists': p.is_file()}
    if p.is_file():
        digest = hashlib.sha256()
        with p.open('rb') as stream:
            for block in iter(lambda: stream.read(4 * 1024 * 1024), b''):
                digest.update(block)
        row.update(bytes=p.stat().st_size, sha256=digest.hexdigest(), matches_expected=digest.hexdigest() == expected)
    record['artifacts'].append(row)
    save()
