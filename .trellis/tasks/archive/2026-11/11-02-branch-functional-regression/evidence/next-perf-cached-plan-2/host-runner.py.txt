def sql(query,db=None):
    result=subprocess.run([PSQL,"-X","-At","-v","ON_ERROR_STOP=1","-c",query],
        env=dict(pg,PGDATABASE=db or pg["PGDATABASE"]),capture_output=True,text=True,timeout=25)
    with (RUN/"database-operations.jsonl").open("a") as f:
        f.write(json.dumps(dict(at=now(),database=db or pg["PGDATABASE"],sql=query,
            exit_code=result.returncode,stdout=result.stdout,stderr=result.stderr))+"\n")
    if result.returncode:
        raise RuntimeError("Database operation failed; inspect task-owned sanitized evidence")
    return result.stdout.strip()
