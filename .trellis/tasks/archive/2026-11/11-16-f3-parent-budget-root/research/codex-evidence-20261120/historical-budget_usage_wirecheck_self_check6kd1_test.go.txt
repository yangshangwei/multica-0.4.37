// Check 6kd1: the checker of budget_usage_wirecheck_check6kd1_test.go is
// only worth something if it refuses what design 11.5 says is refused. This
// file feeds it documents that each break one rule, and a few that break
// none.

const (
	check6kd1Head = `"protocol_version":1,"batch_key":"0195f3a2-0000-7000-8000-000000000002","producer_epoch":"0195f39e-0000-7000-8000-00000000b01d","adapter":"claude","adapter_version":"2.1.34","normalizer_version":"1","reported_at":"2026-11-17T08:00:00Z"`
	check6kd1IDa  = "0195f3a1-0000-7000-8000-00000000000a"
	check6kd1IDb  = "0195f3a1-0000-7000-8000-00000000000b"
)

// check6kd1Doc is a batch with the given envelope fields and invocations.
func check6kd1Doc(head string, invocations ...string) string {
	return `{` + head + `,"invocations":[` + strings.Join(invocations, ",") + `]}`
}

func check6kd1Inv(id, ended string, items ...string) string {
	return `{"invocation_id":"` + id + `","resumed":false,"ended":"` + ended + `","observed_at":"2026-11-17T07:59:58Z","usage":[` + strings.Join(items, ",") + `]}`
}

// check6kd1Item is a usage item; an empty cost leaves cost_ticks out.
func check6kd1Item(model, input, cost string) string {
	item := `{"model":"` + model + `","tokens":{"input":"` + input + `","output":"340","cache_read":"0","cache_write":"0"}`
	if cost != "" {
		item += `,"cost_ticks":"` + cost + `"`
	}
	return item + `}`
}

// check6kd1Example is the example of design 11.5 with complete identifiers.
func check6kd1Example() string {
	return check6kd1Doc(check6kd1Head, check6kd1Inv(check6kd1IDa, "returned", check6kd1Item("claude-fable-5", "1200", "81000000")))
}

// check6kd1Changed replaces one piece of the example and fails when the
// piece is not there exactly once: a case that changes nothing proves
// nothing.
func check6kd1Changed(t *testing.T, old, replacement string) string {
	t.Helper()
	example := check6kd1Example()
	if strings.Count(example, old) != 1 {
		t.Fatalf("the example holds %q %d times, want once", old, strings.Count(example, old))
	}
	return strings.Replace(example, old, replacement, 1)
}

func check6kd1NumberedID(n int) string {
	return fmt.Sprintf("0195f3a1-0000-7000-8000-%012x", n)
}

	for n := range check6kd1MaxInvocations + 1 {
		tooMany = append(tooMany, check6kd1Inv(check6kd1NumberedID(n), "returned"))
	}
	cases = append(cases, refusal{name: "33 invocations", document: check6kd1Doc(check6kd1Head, tooMany...), want: `33 invocations`})
	var crowded []string
	for n := range 9 {
		items := make([]string, 0, 29)
		for m := range 29 {
			items = append(items, check6kd1Item(fmt.Sprintf("model-%02d", m), "1", ""))
		}
		crowded = append(crowded, check6kd1Inv(check6kd1NumberedID(n), "returned", items...))
	}
	cases = append(cases, refusal{name: "261 usage items in 9 invocations", document: check6kd1Doc(check6kd1Head, crowded...), want: `261 usage items`})
	cases = append(cases, refusal{name: "a body over 256 KiB", document: `{` + strings.Repeat(" ", check6kd1MaxBodyBytes) + strings.TrimPrefix(check6kd1Example(), `{`), want: `more than 256 KiB`})

	for _, tc := range cases {
		t.Run(tc.name, func(t *testing.T) {
			t.Parallel()
			document := tc.document
			if tc.old != "" {
				document = check6kd1Changed(t, tc.old, tc.replacement)
			}
			err := check6kd1ValidateBatch([]byte(document))
			if err == nil {
				t.Fatalf("the checker accepts a body that breaks a rule of 11.5:\n%.400s", document)
			}
			if !strings.Contains(err.Error(), tc.want) {
				t.Fatalf("refused for another reason\n got:  %v\n want: ...%s...", err, tc.want)
			}
		})
	}
}

func TestCheck6KD1WireCheckerAccepts(t *testing.T) {
	t.Parallel()

	full := make([]string, 0, check6kd1MaxInvocations)
	for n := range check6kd1MaxInvocations {
		items := make([]string, 0, 8)
		for m := range 8 {
			items = append(items, check6kd1Item(fmt.Sprintf("model-%02d", m), "1", ""))
		}
		full = append(full, check6kd1Inv(check6kd1NumberedID(n), "interrupted", items...))
	}
	cases := []struct {
		name     string
		document string
	}{
		{"the example of 11.5", check6kd1Example()},
		{"no adapter version", strings.Replace(check6kd1Example(), `"adapter_version":"2.1.34",`, ``, 1)},
		{"no invocation", check6kd1Doc(check6kd1Head)},
		{"an unknown invocation without usage", check6kd1Doc(check6kd1Head, check6kd1Inv(check6kd1IDa, "unknown"))},
		{"a returned invocation without usage", check6kd1Doc(check6kd1Head, check6kd1Inv(check6kd1IDa, "returned"))},
		{"an item with four zeros and a cost", check6kd1Doc(check6kd1Head,
			check6kd1Inv(check6kd1IDa, "returned", `{"model":"m","tokens":{"input":"0","output":"0","cache_read":"0","cache_write":"0"},"cost_ticks":"1"}`))},
