package main

import (
	"crypto/x509"
	"os"
)

func init() {
	cert, err := os.ReadFile(os.Getenv("SSL_CERT_FILE"))
	if err != nil {
		panic("missing material fixture CA")
	}
	roots := x509.NewCertPool()
	if !roots.AppendCertsFromPEM(cert) {
		panic("invalid material fixture CA")
	}
	x509.SetFallbackRoots(roots)
}
